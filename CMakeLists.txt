cmake_minimum_required(VERSION 3.20)
project(MiniRaster VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MINIRASTER_UBSAN "Trap undefined behavior in GCC/Clang builds" OFF)

if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(MINGW AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    add_compile_options(-no-canonical-prefixes -fno-canonical-system-headers)
endif()
if(MINIRASTER_UBSAN)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        set(MINIRASTER_SANITIZER_TRAP -fsanitize-trap=undefined)
        if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
            set(MINIRASTER_SANITIZER_TRAP -fsanitize-undefined-trap-on-error)
        endif()
        add_compile_options(-fsanitize=undefined ${MINIRASTER_SANITIZER_TRAP} -fno-omit-frame-pointer)
        add_link_options(-fsanitize=undefined ${MINIRASTER_SANITIZER_TRAP})
    else()
        message(FATAL_ERROR "This UBSan option requires GCC or Clang")
    endif()
endif()

find_package(Threads REQUIRED)
add_library(miniraster_core
    src/clipper.cpp src/framebuffer.cpp src/texture.cpp
    src/thread_pool.cpp src/rasterizer.cpp src/scenes.cpp
)
target_include_directories(miniraster_core PUBLIC include)
target_link_libraries(miniraster_core PUBLIC Threads::Threads)
add_executable(miniraster src/main.cpp)
target_link_libraries(miniraster PRIVATE miniraster_core)
target_compile_definitions(miniraster PRIVATE MINIRASTER_VERSION="${PROJECT_VERSION}")
add_executable(render_bench benchmarks/render_bench.cpp)
target_link_libraries(render_bench PRIVATE miniraster_core)
if(WIN32)
    target_link_libraries(miniraster PRIVATE shell32)
    target_link_libraries(render_bench PRIVATE shell32)
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(test_name math clipper rasterizer thread_pool scenes)
        add_executable(test_${test_name} tests/test_${test_name}.cpp)
        target_link_libraries(test_${test_name} PRIVATE miniraster_core)
        add_test(NAME ${test_name} COMMAND test_${test_name})
        set_tests_properties(${test_name} PROPERTIES TIMEOUT 90)
    endforeach()
    add_test(NAME cli COMMAND "${CMAKE_COMMAND}"
        "-DPROGRAM=$<TARGET_FILE:miniraster>"
        "-DEXPECTED_VERSION=${PROJECT_VERSION}"
        "-DTEST_OUTPUT=${CMAKE_CURRENT_BINARY_DIR}/cli_test.ppm"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.cmake")
    set_tests_properties(cli PROPERTIES TIMEOUT 30)
endif()
